% =====================================
% FORMATO DE TÍTULOS DE CAPÍTULOS
% =====================================

% Capítulos numerados: "Capítulo 1. Título" en la misma línea, centrado, 14pt, negrita
% Con espacio superior de 0.32\textheight (solo para capítulos numerados)
\titleformat{\chapter}[hang]
{\normalfont\bfseries\centering\fontsize{14}{16.8}\selectfont}
{Capítulo \thechapter.}{0.5em}{}

\titlespacing*{\chapter}{0pt}{0.32\textheight}{20pt}

% Capítulos sin número - PEGADOS ARRIBA (0pt antes del título)
% Formato: titlespacing*{comando}{izquierda}{antes}{después}
\titleformat{name=\chapter,numberless}[block]
{\normalfont\bfseries\centering\fontsize{14}{16.8}\selectfont}
{}{0pt}{}

\titlespacing*{name=\chapter,numberless}{0pt}{-50pt}{20pt}

% =====================================
% FORMATO DE SECCIONES Y SUBSECCIONES
% =====================================

% Subcapítulos (secciones): 13pt, negrita, alineado a la izquierda
\titleformat{\section}
{\normalfont\bfseries\fontsize{13}{15.6}\selectfont}
{\thesection}{1em}{}

% Sub-subcapítulos (subsecciones): 12pt, negrita, alineado a la izquierda
\titleformat{\subsection}
{\normalfont\bfseries\fontsize{12}{14.4}\selectfont}
{\thesubsection}{1em}{}
